cmake_minimum_required(VERSION 3.25)
project(FooyinRenumberTracks VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only 64-bit builds are supported.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
if(WIN32)
  find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Sql Test)
else()
  find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets Sql Test)
endif()
set(FOOYIN_SOURCE "" CACHE PATH "fooyin v0.14.1 source tree")
if(WIN32)
  set(FOOYIN_RUNTIME "C:/Program Files/fooyin" CACHE PATH "Matching installed fooyin")
else()
  set(FOOYIN_RUNTIME "" CACHE PATH "Installed fooyin library directory (contains libfooyin_*.so)")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
foreach(module core gui utils)
  string(TOUPPER "fy${module}" macro)
  if(WIN32)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/fy${module}_export.h" CONTENT "#pragma once\n#define ${macro}_EXPORT __declspec(dllimport)\n#define ${macro}_NO_EXPORT\n#define ${macro}_DEPRECATED __declspec(deprecated)\n#define ${macro}_DEPRECATED_EXPORT ${macro}_EXPORT ${macro}_DEPRECATED\n")
    add_library(fooyin_${module} SHARED IMPORTED)
    set_target_properties(fooyin_${module} PROPERTIES IMPORTED_IMPLIB "${CMAKE_SOURCE_DIR}/imports/fooyin_${module}.lib" IMPORTED_LOCATION "${FOOYIN_RUNTIME}/fooyin_${module}.dll")
  else()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/fy${module}_export.h" CONTENT "#pragma once\n#define ${macro}_EXPORT __attribute__((visibility(\"default\")))\n#define ${macro}_NO_EXPORT __attribute__((visibility(\"hidden\")))\n#define ${macro}_DEPRECATED __attribute__((deprecated))\n#define ${macro}_DEPRECATED_EXPORT ${macro}_EXPORT ${macro}_DEPRECATED\n")
    add_library(fooyin_${module} SHARED IMPORTED)
    set_target_properties(fooyin_${module} PROPERTIES IMPORTED_LOCATION "${FOOYIN_RUNTIME}/libfooyin_${module}.so.0.0.0")
  endif()
endforeach()
set(PLUGIN_VERSION "0.1.0")
configure_file(renumber/renumber.json.in generated/renumber.json)
file(GLOB sources CONFIGURE_DEPENDS renumber/*.cpp renumber/*.h)
add_library(fyplugin_renumber MODULE ${sources})
set_target_properties(fyplugin_renumber PROPERTIES PREFIX "")
target_include_directories(fyplugin_renumber PRIVATE "${FOOYIN_SOURCE}/include" "${FOOYIN_SOURCE}/src" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(fyplugin_renumber PRIVATE fooyin_core fooyin_gui fooyin_utils Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Sql)
if(MSVC)
  target_compile_options(fyplugin_renumber PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
else()
  target_compile_options(fyplugin_renumber PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_compile_definitions(fyplugin_renumber PRIVATE QT_NO_KEYWORDS)
enable_testing()
add_executable(plugin_load_test tests/plugin_load_test.cpp)
target_link_libraries(plugin_load_test PRIVATE Qt6::Core)
add_test(NAME plugin_load COMMAND plugin_load_test $<TARGET_FILE:fyplugin_renumber>)
if(WIN32)
  set_tests_properties(plugin_load PROPERTIES ENVIRONMENT "QT_PLUGIN_PATH=${FOOYIN_RUNTIME}")
else()
  set_tests_properties(plugin_load PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${FOOYIN_RUNTIME}:$ENV{LD_LIBRARY_PATH}")
endif()
